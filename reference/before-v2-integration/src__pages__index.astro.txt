const {pieces,edges}=await getPortfolio();
const jobs=pieces.filter(p=>p.data.kind==='job').sort((a,b)=>{
 const rank=(id:string)=>{const i=site.resumeOrder.indexOf(id);return i<0?Infinity:i;};
 return rank(a.id)-rank(b.id)||a.data.title.localeCompare(b.data.title);
});
const jobIds=new Set(jobs.map(j=>j.id));
// Nest a subject under its first declared Job context; show each subject once.
const parentFor=new Map<string,string>();
for(const edge of edges)if(edge.type==='part-of'&&jobIds.has(edge.target)&&!jobIds.has(edge.source)&&!parentFor.has(edge.source))parentFor.set(edge.source,edge.target);
const independent=pieces.filter(p=>p.data.kind!=='job'&&!parentFor.has(p.id));
const topics=[...new Set(pieces.flatMap(p=>p.data.topics??[]))].sort((a,b)=>a.localeCompare(b));
const filterData=pieces.map(p=>({id:p.id,kind:p.data.kind,topics:p.data.topics??[],search:[p.data.title,p.data.label,p.data.summary,p.data.role,p.data.status,p.data.attribution,...(p.data.topics??[])].filter(Boolean).join(' ')}));
const contexts=edges.filter(e=>e.type==='part-of'&&jobIds.has(e.target)&&!jobIds.has(e.source)).map(e=>({child:e.source,parent:e.target}));

<script>
 import {portfolioView,type FilterNode,type ContextLink,type FocusEdge} from '../lib/filters';
 const root=document.querySelector<HTMLElement>('[data-portfolio]');
 const element=document.querySelector('#portfolio-data');
 if(root&&element){
  const data=JSON.parse(element.textContent!) as {nodes:FilterNode[];contexts:ContextLink[];edges:FocusEdge[]};
  let kind='all',topic='all',selected:string|null=null;
  const names=new Map(data.nodes.map(node=>[node.id,document.querySelector(`[data-id="${node.id}"]`)?.getAttribute('aria-label')??node.id]));
  function update(){
   const view=portfolioView(data.nodes,data.contexts,data.edges,{query:'',kind,topic},selected);selected=view.selected;
   root!.querySelectorAll<HTMLElement>('[data-resume-id]').forEach(entry=>{
    const id=entry.dataset.resumeId!;entry.hidden=!view.focus.has(id);entry.classList.toggle('is-context',view.contextOnly.has(id));entry.classList.toggle('is-focused',id===selected);
    const note=entry.querySelector<HTMLElement>('.resume-context-note');if(note)note.hidden=!view.contextOnly.has(id);
   });
   root!.querySelectorAll<HTMLElement>('[data-job-group], [data-resume-group]').forEach(group=>{group.hidden=!Array.from(group.querySelectorAll<HTMLElement>('[data-resume-id]')).some(e=>!e.hidden);});
   root!.querySelector<HTMLElement>('[data-resume-empty]')!.hidden=view.focus.size>0;
   root!.querySelectorAll<HTMLButtonElement>('[data-kind]').forEach(button=>button.setAttribute('aria-pressed',String(button.dataset.kind===kind)));
   root!.querySelectorAll<HTMLButtonElement>('[data-topic]').forEach(button=>button.setAttribute('aria-pressed',String(button.dataset.topic===topic)));
   root!.querySelectorAll<HTMLButtonElement>('[data-focus-id]').forEach(button=>button.setAttribute('aria-pressed',String(button.dataset.focusId===selected)));
   root!.querySelector<HTMLElement>('[data-clear-focus]')!.hidden=!selected;
   root!.querySelector<HTMLElement>('[data-reset-filters]')!.hidden=kind==='all'&&topic==='all'&&!selected;
   root!.querySelector('[data-focus-description]')!.textContent=selected?`Focused on ${names.get(selected)} and its connections.`:'Select a subject in either view to follow its connections.';
   root!.querySelector('[data-filter-status]')!.textContent=selected?`${view.focus.size} ${view.focus.size===1?'subject':'subjects'} in focus · résumé and map connected`:`${view.visible.size} ${view.visible.size===1?'subject':'subjects'} · résumé and map connected`;
   document.dispatchEvent(new CustomEvent('portfolio:filter',{detail:{visibleIds:[...view.visible],contextIds:[...view.contextOnly],focusIds:[...view.focus],selectedId:selected}}));
  }
  root.querySelectorAll<HTMLButtonElement>('[data-kind]').forEach(button=>button.addEventListener('click',()=>{kind=button.dataset.kind!;update();}));
  root.querySelectorAll<HTMLButtonElement>('[data-topic]').forEach(button=>button.addEventListener('click',()=>{topic=topic===button.dataset.topic?'all':button.dataset.topic!;update();}));
  root.querySelector('[data-clear-focus]')!.addEventListener('click',()=>{selected=null;update();});
  root.querySelector('[data-reset-filters]')!.addEventListener('click',()=>{kind='all';topic='all';selected=null;update();});
  root.querySelectorAll<HTMLButtonElement>('[data-focus-id]').forEach(button=>button.addEventListener('click',()=>{selected=selected===button.dataset.focusId?null:button.dataset.focusId!;update();}));
  document.addEventListener('portfolio:select',event=>{
   const map=document.querySelector('#map')!;const before=map.getBoundingClientRect();
   selected=(event as CustomEvent<{id:string|null}>).detail.id;update();
   // Keep the map in place while résumé rows above it change height.
   if(before.top<innerHeight&&before.bottom>0)requestAnimationFrame(()=>window.scrollBy({top:map.getBoundingClientRect().top-before.top,behavior:'instant'}));
  });
  root.querySelector<HTMLElement>('.shared-filters')!.hidden=false;
  document.addEventListener('portfolio:ready',update);update();
 }
</script>
